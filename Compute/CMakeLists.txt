cmake_minimum_required(VERSION 4.2.3)
project(Ir77Compute LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB_RECURSE Ir77Compute_SOURCES
    "*.cpp"
)

add_library(Ir77Compute STATIC ${Ir77Compute_SOURCES})

# the kernel's text as a C++ string -- kernels/Ir77Integrals.cl (built by OpenCL at runtime). Regenerated whenever it changes (CMake re-runs
# on an edit).
set(Ir77Compute_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(Ir77Compute_EMBED
    COMPUTE_KERNEL_INTEGRALS ${CMAKE_CURRENT_SOURCE_DIR}/kernels/Ir77Integrals.cl
)

set(Ir77Compute_HEADER "#pragma once\n\n// GENERATED by Compute/CMakeLists.txt -- edit the files named below, not this.\n\nnamespace NSIr77Compute {\n")
list(LENGTH Ir77Compute_EMBED Ir77Compute_EMBED_LENGTH)
math(EXPR Ir77Compute_EMBED_LAST "${Ir77Compute_EMBED_LENGTH} - 1")
foreach(index RANGE 0 ${Ir77Compute_EMBED_LAST} 2)
    math(EXPR file_index "${index} + 1")
    list(GET Ir77Compute_EMBED ${index} name)
    list(GET Ir77Compute_EMBED ${file_index} path)
    file(READ ${path} text)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${path})
    string(APPEND Ir77Compute_HEADER "\n// ${path}\ninline constexpr char const* ${name}{R\"IR77EMBED(${text})IR77EMBED\"};\n")
endforeach()
string(APPEND Ir77Compute_HEADER "\n}  // namespace NSIr77Compute\n")
file(CONFIGURE OUTPUT ${Ir77Compute_GENERATED}/Ir77ComputeSources.hpp CONTENT "${Ir77Compute_HEADER}" @ONLY)

target_include_directories(Ir77Compute PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/dictionary
    ${CMAKE_CURRENT_SOURCE_DIR}/interface
    ${CMAKE_CURRENT_SOURCE_DIR}/runtime
    ${CMAKE_CURRENT_SOURCE_DIR}/server
    ${CMAKE_CURRENT_SOURCE_DIR}/services
    ${CMAKE_CURRENT_SOURCE_DIR}/types
    ${Ir77Compute_GENERATED}
)

target_compile_options(Ir77Compute PRIVATE
    -fno-implicit-modules
    -fno-implicit-module-maps
)

# OpenCL -- the SDK when it's installed; otherwise Khronos' headers and ICD loader, fetched (the GPU driver supplies the actual device)
find_package(OpenCL QUIET)
if(NOT OpenCL_FOUND)
    include(FetchContent)

    FetchContent_Declare(
        OpenCLHeaders
        GIT_REPOSITORY https://github.com/KhronosGroup/OpenCL-Headers.git
        GIT_TAG        v2025.07.22
    )

    FetchContent_Declare(
        OpenCLICDLoader
        GIT_REPOSITORY https://github.com/KhronosGroup/OpenCL-ICD-Loader.git
        GIT_TAG        v2025.07.22
    )

    # the headers first -- the loader builds against them
    FetchContent_MakeAvailable(OpenCLHeaders)
    set(OPENCL_ICD_LOADER_HEADERS_DIR ${openclheaders_SOURCE_DIR} CACHE PATH "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(OpenCLICDLoader)
endif()

target_compile_definitions(Ir77Compute PUBLIC CL_TARGET_OPENCL_VERSION=120)
target_link_libraries(Ir77Compute PUBLIC OpenCL::OpenCL)

target_link_libraries(Ir77Compute PUBLIC Ir77RT)

# the pipe's JSON -- the simdjson::simdjson target is PeregrineV's fetch (configured before Compute)
target_link_libraries(Ir77Compute PUBLIC simdjson::simdjson)
